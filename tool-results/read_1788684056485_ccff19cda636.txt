   156→        e.preventDefault()

   158→      } else if (mod && e.key === 'k') {
   159→        e.preventDefault()
   160→        setShortcutsOpen((v) => !v)
   161→      }
   162→    }

   662→                        icon={<ScrollText className="h-8 w-8" />}
